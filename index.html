const express = require('express'); const path = require('path'); const app = express(); const PORT = process.env.PORT || 3000; // JSON 및 Form 데이터 파싱 미들웨어 app.use(express.json()); app.use(express.urlencoded({ extended: true })); // 정적 파일(HTML, CSS, JS) 제공 app.use(express.static(path.join(__dirname, 'public'))); // 메인 페이지 라우트 app.get('/', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'index.html')); }); // 테스트용 데이터 API (로그인 없이 누구나 접근 가능) app.get('/api/data', (req, res) => { res.json({ message: "로그인 없이 이용 가능한 서비스입니다.", items: ["아이템 1", "아이템 2", "아이템 3"] }); }); // 서버 실행 app.listen(PORT, () => { console.log(`서버가 http://localhost:${PORT} 에서 실행 중입니다.`); });
